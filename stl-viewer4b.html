<script type="module">
import * as THREE from 'three';
import { OrbitControls } from
'three/addons/controls/OrbitControls.js';
import { STLLoader } from
'three/addons/loaders/STLLoader.js';
// Scene setup
const container = document.getElementById('viewer');
const scene = new THREE.Scene();
scene.background = new THREE.Color(0xe8edf2);
const camera = new THREE.PerspectiveCamera(
45,
container.clientWidth / container.clientHeight,
0.1,
100000
);
camera.up.set(0, 0, 1);
camera.position.set(150, 150, 200);
const renderer = new THREE.WebGLRenderer({
antialias: true
});
renderer.setPixelRatio(
Math.min(window.devicePixelRatio, 2)
);
renderer.setSize(
container.clientWidth,
container.clientHeight
);
renderer.outputColorSpace = THREE.SRGBColorSpace;
container.appendChild(renderer.domElement);
// Lighting
scene.add(new THREE.AmbientLight(0xffffff, 1.2));
const light1 = new THREE.DirectionalLight(0xffffff, 2);
light1.position.set(100, 200, 150);
scene.add(light1);
const light2 = new THREE.DirectionalLight(0xffffff, 1);
light2.position.set(-100, -100, -100);
scene.add(light2);
// Orbit controls
const controls = new OrbitControls(
camera,
renderer.domElement
);
controls.enableDamping = true;
controls.dampingFactor = 0.06;
controls.enableZoom = true;
controls.enablePan = true;
controls.rotateSpeed = 0.8;
// Grid
const grid = null;
// Model
let currentModel = null;
let originalCameraPosition = new THREE.Vector3(150, 150, 200);
let originalTarget = new THREE.Vector3(0, 0, 0);
const loader = new STLLoader();
const material = new THREE.MeshStandardMaterial({
color: 0xcccccc,
metalness: 0.2,
roughness: 0.55
});
// Load STL geometry
function loadSTL(geometry, name) {
geometry.computeVertexNormals();
geometry.computeBoundingBox();
if (currentModel) {
scene.remove(currentModel);
currentModel.geometry.dispose();
}
const model = new THREE.Mesh(
geometry,
material.clone()
);
// Centre model at origin
const centre = new THREE.Vector3();
geometry.boundingBox.getCenter(centre);
model.position.sub(centre);
currentModel = model;
scene.add(currentModel);
// Automatically position camera
const size = new THREE.Vector3();
geometry.boundingBox.getSize(size);
const maxDimension = Math.max(
size.x,
size.y,
size.z,
1
);
const distance = maxDimension * 0.5;
camera.position.set(
distance,
-distance * 7,
0
);
camera.near = maxDimension / 1000;
camera.far = maxDimension * 100;
camera.updateProjectionMatrix();
controls.target.set(0, 0, 0);
controls.update();
originalCameraPosition.copy(camera.position);
originalTarget.set(0, 0, 0);
grid.scale.setScalar(
maxDimension / 200
);
document.getElementById('filename').textContent =
name;
}
// File input
document.getElementById('fileInput')
.addEventListener('change', function(event) {
const file = event.target.files[0];
if (!file) return;
const reader = new FileReader();
reader.onload = function(e) {
const geometry = loader.parse(e.target.result);
loadSTL(geometry, file.name);
};
reader.readAsArrayBuffer(file);
});
// Load STL from URL
function loadFromURL(url) {
loader.load(
url,
function(geometry) {
loadSTL(geometry, 'STL model');
},
undefined,
function(error) {
console.error('STL loading error:', error);
document.getElementById('filename').textContent =
'Could not load model';
}
);
}
// Automatically load STL from iframe URL parameter
const params = new URLSearchParams(window.location.search);
const modelURL = params.get('model');
if (modelURL) {
loadFromURL(modelURL);
}
// Reset camera
document.getElementById('resetButton')
.addEventListener('click', function() {
camera.position.copy(originalCameraPosition);
controls.target.copy(originalTarget);
controls.update();
});
// Wireframe toggle
let wireframe = false;
document.getElementById('wireButton')
.addEventListener('click', function() {
if (!currentModel) return;
wireframe = !wireframe;
currentModel.material.wireframe = wireframe;
this.textContent = wireframe
? 'Solid View'
: 'Wireframe';
});
// Drag and drop
const dropMessage = document.getElementById('dropMessage');
container.addEventListener('dragover', function(e) {
e.preventDefault();
dropMessage.style.display = 'flex';
});
container.addEventListener('dragleave', function(e) {
if (!container.contains(e.relatedTarget)) {
dropMessage.style.display = 'none';
}
});
container.addEventListener('drop', function(e) {
e.preventDefault();
dropMessage.style.display = 'none';
const file = e.dataTransfer.files[0];
if (!file || !file.name.toLowerCase().endsWith('.stl')) {
return;
}
const reader = new FileReader();
reader.onload = function(event) {
const geometry = loader.parse(event.target.result);
loadSTL(geometry, file.name);
};
reader.readAsArrayBuffer(file);
});
// Responsive resizing
window.addEventListener('resize', function() {
camera.aspect =
container.clientWidth / container.clientHeight;
camera.updateProjectionMatrix();
renderer.setSize(
container.clientWidth,
container.clientHeight
);
});
// Animation loop
function animate() {
requestAnimationFrame(animate);
controls.update();
renderer.render(scene, camera);
}
animate();
